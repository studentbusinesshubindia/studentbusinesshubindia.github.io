function doPost(e) { var lock = LockService.getScriptLock(); // एकाच वेळी अनेक युजर्स फॉर्म भरत असतील तर Lock मिळवण्यासाठी १० सेकंद थांबेल lock.tryLock(10000); try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // -------------------------------------------------- // 📥 ०. Request Data Parse (JSON & Form Parameter Support) // -------------------------------------------------- var p = {}; if (e && e.postData && e.postData.contents) { try { p = JSON.parse(e.postData.contents); } catch (err) { p = e.parameter || {}; } } else if (e && e.parameter) { p = e.parameter; } var action = p.action || ""; // ========================================== // 🔐 १. लॉगिन व्हॅलिडेशन (VERIFY_LOGIN) // ========================================== if (action === "VERIFY_LOGIN") { var custSheetForVerify = ss.getSheetByName("Customers"); if (!custSheetForVerify) { return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: "Customers Sheet सापडली नाही!" })) .setMimeType(ContentService.MimeType.JSON); } var data = custSheetForVerify.getDataRange().getValues(); var inputCustId = p.customerId || p.loginCustomerId || ""; var inputPassword = p.password || p.loginPassword || ""; var userFound = false; var isPasswordCorrect = false; for (var i = 1; i < data.length; i++) { var rowCustId = data[i][0]; // Column A: Customer ID var rowPassword = data[i][14]; // Column O: Password (१५ वा कॉलम) if (rowCustId && rowCustId.toString().trim() === inputCustId.toString().trim()) { userFound = true; if (rowPassword && rowPassword.toString().trim() === inputPassword.toString().trim()) { isPasswordCorrect = true; break; } } } if (!userFound) { return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: "हा Customer ID सापडला नाही!" })) .setMimeType(ContentService.MimeType.JSON); } else if (!isPasswordCorrect) { return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: "चुकीचा पासवर्ड! कृपया बरोबर पासवर्ड टाका." })) .setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", message: "लॉगिन यशस्वी!" })) .setMimeType(ContentService.MimeType.JSON); } } // ========================================== // 📝 २. STEP 1 REGISTRATION (नवीन Serial ID जनरेट करणे) // ========================================== if (action === "STEP_1_REGISTRATION") { var mainSheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0]; var custSheet = ss.getSheetByName("Customers") || ss.insertSheet("Customers"); // अनुक्रमाने कस्टमर आयडी शोधणे (CUST1001, CUST1002...) var custValues = custSheet.getDataRange().getValues(); var nextIdNum = 1001; for (var k = 1; k < custValues.length; k++) { var existingId = custValues[k][0] ? custValues[k][0].toString().trim() : ""; var match = existingId.match(/CUST(\d+)/); if (match) { var idNum = parseInt(match[1], 10); if (idNum >= nextIdNum) { nextIdNum = idNum + 1; } } } var generatedCustId = "CUST" + nextIdNum; if (mainSheet) { var catVal = p.init_category || p.category || p.interest || "-"; mainSheet.appendRow([ new Date(), p.type || "Business Owner", p.name || p.init_name || "-", p.businessName || p.init_business || "-", p.contact || p.init_mobile || "-", p.city || p.init_city || "-", catVal ]); } return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", result: "success", generatedCustomerId: generatedCustId, customerId: generatedCustId })).setMimeType(ContentService.MimeType.JSON); } // ========================================== // 👥 ३. CUSTOMERS SHEET (Duplicate Check & Save/Update) // ========================================== var custSheet = ss.getSheetByName("Customers") || ss.insertSheet("Customers"); var sharedCustId = p.customerId || p.loginCustomerId || p.custId || ""; var incomingMobile = p.custMobile || p.init_mobile || p.mobile || ""; var panVal = p.panCard || p.panNo || p.pan || p.pan_card || p.pancard || "-"; var emailVal = p.custEmail || p.email || p.init_email || p.emailId || "-"; var pinVal = p.custPincode || p.pincode || p.pinCode || p.init_pincode || p.pin || "-"; var cityVal = p.custCity || p.city || p.init_city || "-"; var stateVal = p.custState || p.state || p.init_state || "-"; var addrVal = p.custAddress || p.address || p.init_address || "-"; var passVal = p.password || p.pass || p.custPassword || p.userPassword || "-"; var custData = custSheet.getDataRange().getValues(); var existingRowIndex = -1; // 🔍 Customer ID किंवा Mobile ने ओळ शोधणे for (var r = 1; r < custData.length; r++) { var rowCustId = custData[r][0] ? custData[r][0].toString().trim() : ""; var rowMobile = custData[r][4] ? custData[r][4].toString().trim() : ""; var matchByCustId = (sharedCustId && rowCustId === sharedCustId.toString().trim()); var matchByMobile = (incomingMobile && rowMobile === incomingMobile.toString().trim()); if (matchByCustId || matchByMobile) { existingRowIndex = r + 1; sharedCustId = rowCustId || sharedCustId; break; } } // जर ID नसेल तर नवीन CUST ID जनरेट करणे if (!sharedCustId || sharedCustId.toString().trim() === "") { var maxNum = 1000; for (var c = 1; c < custData.length; c++) { var cId = custData[c][0] ? custData[c][0].toString().trim() : ""; var m = cId.match(/CUST(\d+)/); if (m) { var num = parseInt(m[1], 10); if (num > maxNum) maxNum = num; } } sharedCustId = "CUST" + (maxNum + 1); } else { sharedCustId = sharedCustId.toString().trim(); } var rowData = [ sharedCustId, // A: Customer ID p.custName || p.init_name || p.ownerName || "-", // B: Name p.businessName || p.init_business || "-", // C: Business Name p.shopName || p.businessName || "-", // D: Shop Name p.custMobile || p.init_mobile || p.mobile || "-", // E: Mobile emailVal, // F: Email ID panVal, // G: PAN Card p.gstNo || p.gst || p.gstin || "-", // H: GST No stateVal, // I: State cityVal, // J: City addrVal, // K: Address pinVal, // L: Pin Code p.employeeId || p.empId || p.empCode || "-", // M: Employee ID new Date(), // N: Registered Date passVal // O: Password ]; if (existingRowIndex > 0) { custSheet.getRange(existingRowIndex, 1, 1, rowData.length).setValues([rowData]); } else { custSheet.appendRow(rowData); } // ========================================== // 📄 ४. SHEET1 (सामान्य नोंदणी) // ========================================== if (action !== "STEP_1_REGISTRATION") { var mainSheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0]; if (mainSheet) { var catVal = p.init_category || p.businessCategory || p.category || p.bizCategory || p.interest || "-"; mainSheet.appendRow([ new Date(), "Business Owner", p.custName || p.init_name || p.ownerName || p.name || "-", p.businessName || p.init_business || p.bName || "-", p.custMobile || p.init_mobile || p.mobile || p.contact || "-", p.custCity || p.city || p.init_city || p.ownerCity || "-", catVal ]); } } // ========================================== // 📦 ५. PRODUCTS SHEET (प्रॉडक्ट्स सेव्ह करणे) // ========================================== var prodSheet = ss.getSheetByName("Products") || ss.insertSheet("Products"); if (prodSheet) { // हेडर रो नसल्यास तयार करा if (prodSheet.getLastRow() === 0) { prodSheet.appendRow(["Customer ID", "Employee ID", "Product Name", "MRP", "Selling Price", "Quantity", "Drive Link", "Added Date", "Display Image"]); } if (p.productsJson || p.products) { try { var rawProds = p.productsJson || p.products; var prods = typeof rawProds === 'string' ? JSON.parse(rawProds) : rawProds; if (Array.isArray(prods) && prods.length > 0) { prods.forEach(function(item) { saveSingleProduct(prodSheet, p, item, sharedCustId); }); } else { saveSingleProduct(prodSheet, p, p, sharedCustId); } } catch(err) { saveSingleProduct(prodSheet, p, p, sharedCustId); } } else { saveSingleProduct(prodSheet, p, p, sharedCustId); } } var jsonResponse = { status: "SUCCESS", result: "success", message: "Data saved successfully", customerId: sharedCustId }; return ContentService.createTextOutput(JSON.stringify(jsonResponse)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { var errorResponse = { status: "ERROR", result: "error", message: err.toString() }; return ContentService.createTextOutput(JSON.stringify(errorResponse)) .setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } } // ========================================== // 💡 doGet फंक्शन (वेबसाईटवरून GET रिक्वेस्ट आल्यास) // ========================================== function doGet(e) { return doPost(e); } // ========================================== // 🛍️ ६. SINGLE PRODUCT SAVE HELPER FUNCTION // ========================================== function saveSingleProduct(sheet, p, item, sharedCustId) { var pName = item.prodName || item.productName || item.name || item.prod_name || p.prodName || p.productName || p.name || ""; var rawImg = item.rawImg || item.image || item.currentImageBase64 || item.imgData || p.rawImg || p.currentImageBase64 || p.image || ""; if (!pName && !rawImg) return; var driveUrl = "फोटो नाही"; var imageFormula = "फोटो नाही"; if (rawImg && rawImg.length > 50) { var imgInfo = saveImageToDrive(rawImg, pName || "Product"); driveUrl = imgInfo.url; if (imgInfo.directUrl && imgInfo.directUrl !== "") { imageFormula = '=IMAGE("' + imgInfo.directUrl + '")'; } else { imageFormula = driveUrl; } } sheet.appendRow([ sharedCustId, // A: Customer ID item.employeeId || item.empId || p.employeeId || p.empId || "-", // B: Employee ID pName || "Product", // C: Product Name item.mrp || item.prodPrice || p.mrp || p.prodPrice || 0, // D: MRP item.sellingPrice || item.price || p.sellingPrice || p.price || 0, // E: Selling Price item.quantity || item.qty || p.quantity || 1, // F: Quantity driveUrl, // G: Drive Link new Date(), // H: Added Date imageFormula // I: Display Image ]); } // ========================================== // 🖼️ ७. DRIVE IMAGE SAVE HELPER FUNCTION // ========================================== function saveImageToDrive(base64Data, fileName) { try { if (!base64Data || typeof base64Data !== 'string') { return { url: "फोटो नाही", directUrl: "" }; } var pureBase64 = base64Data; var contentType = "image/jpeg"; if (pureBase64.indexOf('%') !== -1) { pureBase64 = decodeURIComponent(pureBase64); } pureBase64 = pureBase64.replace(/ /g, '+'); if (pureBase64.indexOf('base64,') !== -1) { var parts = pureBase64.split('base64,'); var header = parts[0]; pureBase64 = parts[1]; if (header.indexOf('data:') !== -1 && header.indexOf(';') !== -1) { contentType = header.substring(header.indexOf('data:') + 5, header.indexOf(';')); } } pureBase64 = pureBase64.replace(/[^A-Za-z0-9\+\/\=]/g, ""); if (!pureBase64 || pureBase64.length === 0) { return { url: "Upload Error: Empty Image Data", directUrl: "" }; } var decodedData = Utilities.base64Decode(pureBase64); var cleanFileName = (fileName || "Product").replace(/[^a-zA-Z0-9_-]/g, "_"); var blob = Utilities.newBlob(decodedData, contentType, cleanFileName + "_" + new Date().getTime() + ".jpg"); var folder = DriveApp.getRootFolder(); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { url: file.getUrl(), directUrl: "https://lh3.googleusercontent.com/d/" + file.getId() }; } catch(e) { return { url: "Upload Error: " + e.toString(), directUrl: "" }; } }
